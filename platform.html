/* KHAFAIA MVP: email accounts and persistent marketplace posts. */ (() => { const url = window.KHAFAIA_SUPABASE_URL || ''; const key = window.KHAFAIA_SUPABASE_PUBLISHABLE_KEY || ''; const configured = url.startsWith('https://') && !url.includes('YOUR-PROJECT') && key.startsWith('sb_publishable_') && !key.includes('REPLACE_WITH'); const $ = (selector) => document.querySelector(selector); const state = window.khafaiaDemo; let db = null; let user = null; let postKind = 'supplier_offer'; let pendingPostKind = null; let authMode = 'signin'; const authBg = $('#authBg'); const authForm = $('#authForm'); const authMessage = $('#authMessage'); const authButton = $('#authButton'); function notify(message) { if (state?.showToast) state.showToast(message); else window.alert(message); } function setAuthMessage(message, isError = false) { authMessage.textContent = message; authMessage.style.color = isError ? Please try again.', true); } finally { $('#authSubmit').disabled = false; } } async function handlePostSubmit(event) { event.preventDefault(); if (!user) { notify('Sign in before posting.'); openAuth(); return; } const form = $('#listingForm'); const submit = form.querySelector('[type="submit"]'); const post = { post_kind: postKind, category: $('#newType').value, species: $('#postSpecies').value.trim() || null, quantity: Number($('#postQuantity').value), unit: $('#postUnit').value, governorate: $('#postLocation').value, material_state: $('#postCondition').value, moisture: $('#postMoisture').value, availability: $('#postAvailability').value.trim() || null, preparation: $('#postPreparation').value.trim() || null, intended_application: $('#postApplication').value.trim() || null, }; submit.disabled = true; submit.textContent = 'Saving…'; const { error } = await db.from('material_posts').insert(post); submit.disabled = false; submit.textContent = postKind === 'buyer_need' ? Use governorate-level location only; do not add phone numbers, emails, or private details.'; $('#listingForm button[type=submit]').textContent = kind === 'buyer_need' ? 'signup' : 'signin'; $('#authTitle').textContent = authMode === 'signup' ? 'Create your KHAFAIA account' : 'Sign in to KHAFAIA'; $('#authSubmit').textContent = authMode === 'signup' ? 'Create account' : 'Sign in'; $('#authToggle').textContent = authMode === 'signup' ? 'I already have an account' : 'Create an account'; setAuthMessage(authMode === 'signup' ? 'Email confirmation may be required before your first sign-in.' : 'Use your email to manage your palm-material posts.'); }); $('#listingForm').onsubmit = null; $('#listMaterial').onclick = null; $('#buyerNeed').onclick = null; authForm.addEventListener('submit', handleAuthSubmit); $('#listingForm').addEventListener('submit', handlePostSubmit); $('#listMaterial').addEventListener('click', (event) => { event.preventDefault(); openPostForm('supplier_offer'); }); $('#buyerNeed').addEventListener('click', (event) => { event.preventDefault(); openPostForm('buyer_need'); }); refreshAccountButton(); if (!configured || !window.supabase?.createClient) { console.info('KHAFAIA is in local demo mode; configure Supabase to enable accounts and saved posts.'); return; } db = window.supabase.createClient(url, key); db.auth.getSession().then(({ data }) => { user = data.session?.user || null; refreshAccountButton(); loadPosts(); }); db.auth.onAuthStateChange((_event, session) => { user = session?.user || null; refreshAccountButton(); loadPosts(); }); } window.khafaiaPlatform = { deletePost: async (id) => { if (!db || !user || !window.confirm('Remove this marketplace post?')) return; const { error } = await db.from('material_posts').delete().eq('id', id); if (error) notify(`Could not remove the post: ${error.message}`); else { notify('Post removed.'); await loadPosts(); } } }; boot(); })();
